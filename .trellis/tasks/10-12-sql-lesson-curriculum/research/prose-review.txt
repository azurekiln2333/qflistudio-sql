#1 [basics/beginner/SQLite 3.50] SELECT 基础与列选择
objective: 用 SELECT 指定要返回的列，并给列起别名
-- setup --
CREATE TABLE employees (id INTEGER PRIMARY KEY, name TEXT NOT NULL, salary REAL NOT NULL);
INSERT INTO employees (id, name, salary) VALUES
  (1, 'Alice', 12000.0),
  (2, 'Bob', 9000.0),
  (3, 'Carol', 15000.0);
-- query --
SELECT name AS 姓名, salary AS 月薪 FROM employees ORDER BY salary DESC;
-- expected --
Carol|15000.0
Alice|12000.0
Bob|9000.0
-- explain --
SQLite 3.50 中 SELECT 的基本形态是 SELECT 列表 FROM 表。SELECT 后面的列表决定返回哪些列，顺序也由它决定，与建表顺序无关。AS 给列起别名，别名只影响结果集的列名，不改变存储。示例先建一张 employees 表并插入三行，然后只投影 name 和 salary 两列并按 salary 降序，因此返回的是薪资从高到低的三行；这里没有 WHERE，全部行都参与。
-- pitfalls --
  * SELECT * 会随表结构变化而变化，生产查询应显式列出所需列。
  * 表达式列的别名在 WHERE/GROUP BY 中的可见性在 SQLite 与 MySQL 有差异。
-- exercises --
  (1) 只返回 name 列，并按 name 升序。 | hint=SELECT 指定单列后接 ORDER BY。
      answer: SELECT name FROM employees ORDER BY name;
      expected: Alice
Bob
Carol
  (2) 返回 id 与 name 两列，按 id 升序。 | hint=列顺序由 SELECT 列表决定。
      answer: SELECT id, name FROM employees ORDER BY id;
      expected: 1|Alice
2|Bob
3|Carol

#2 [basics/beginner/SQLite 3.50] WHERE 行过滤入门
objective: 用 WHERE 和比较运算符筛选满足条件的行
-- setup --
CREATE TABLE products (id INTEGER PRIMARY KEY, name TEXT NOT NULL, price REAL NOT NULL);
INSERT INTO products VALUES
  (1, '键盘', 199.0),
  (2, '鼠标', 89.0),
  (3, '显示器', 899.0),
  (4, 'U盘', 100.0);
-- query --
SELECT id, name, price FROM products WHERE price > 100 ORDER BY id;
-- expected --
1|键盘|199.0
3|显示器|899.0
-- explain --
WHERE 在 FROM 之后、返回之前对每一行求值，只有条件为真的行进入结果。SQLite 3.50 支持 =、<>（或 !=）、<、<=、>、>= 等比较运算符。示例在 products 表上用price > 100 过滤，返回价格严格大于 100 的商品：键盘 199 与显示器 899 命中，鼠标 89 因低于 100 被排除，U盘恰为 100 既不大于也不被选中——这正是“严格大于”与“大于等于”的分界，若改成 price >= 100 则 U盘 会进入结果。结果按 id 排序以保证可复现。
-- pitfalls --
  * NULL 与任何比较都返回 NULL（非真），需要用 IS NULL 判断。
  * 比较字符串时大小写与排序规则由 collation 决定。
-- exercises --
  (1) 返回价格低于 100 的商品 id 与 name，按 id 排序。 | hint=使用 price < 100。
      answer: SELECT id, name FROM products WHERE price < 100 ORDER BY id;
      expected: 2|鼠标
  (2) 返回名称不是“鼠标”的商品 id，按 id 排序。 | hint=使用 <> 或 !=。
      answer: SELECT id FROM products WHERE name <> '鼠标' ORDER BY id;
      expected: 1
3
4

#3 [basics/beginner/SQLite 3.50] 逻辑运算符与优先级
objective: 组合多个条件并用括号控制优先级
-- setup --
CREATE TABLE staff (id INTEGER PRIMARY KEY, name TEXT, department TEXT, active INTEGER);
INSERT INTO staff VALUES
  (1, 'Alice', 'sales', 1),
  (2, 'Bob', 'sales', 0),
  (3, 'Carol', 'support', 1),
  (4, 'Dave', 'engineering', 1);
-- query --
SELECT id, name FROM staff WHERE (department = 'sales' OR department = 'support') AND active = 1 ORDER BY id;
-- expected --
1|Alice
3|Carol
-- explain --
AND 比 OR 优先级高，所以 a OR b AND c 等价于 a OR (b AND c)。要表达“两个备选之一再叠加一个条件”必须显式加括号。SQLite 3.50 中 AND/OR/NOT 遵循三值逻辑。示例用 (department='sales' OR department='support') AND active=1 选出在职的销售或支持人员，括号确保 OR 先分组再看 active。
-- pitfalls --
  * 漏写括号时 OR 会被整体条件优先级影响，结果常与预期不同。
  * NOT 作用于整个后续表达式时也需要括号界定范围。
-- exercises --
  (1) 返回工程部门或非在职的员工 id，按 id 排序。 | hint=用 department='engineering' OR active=0。
      answer: SELECT id FROM staff WHERE department = 'engineering' OR active = 0 ORDER BY id;
      expected: 2
4
  (2) 返回既不是销售又不在职的员工 id，按 id 排序。 | hint=用 NOT 或 <> 配 AND。
      answer: SELECT id FROM staff WHERE department <> 'sales' AND active = 0 ORDER BY id;
      expected: (no rows)

#4 [basics/beginner/SQLite 3.50] BETWEEN 与 IN 区间判断
objective: 用 BETWEEN 表示闭区间、用 IN 表示集合成员
-- setup --
CREATE TABLE scores (id INTEGER PRIMARY KEY, name TEXT, score INTEGER);
INSERT INTO scores VALUES
  (1, 'Alice', 75),
  (2, 'Bob', 80),
  (3, 'Carol', 59),
  (4, 'Dave', 60),
  (5, 'Eve', 95);
-- query --
SELECT name, score FROM scores WHERE score BETWEEN 60 AND 80 ORDER BY score, name;
-- expected --
Dave|60
Alice|75
Bob|80
-- explain --
BETWEEN a AND b 是闭区间，等价于 x >= a AND x <= b，两端都包含。IN (…) 判断值是否属于给定集合，等价于一串 OR，但更短更清晰。示例用 score BETWEEN 60 AND 80 选出 60 到 80 分（含两端）的学生，因此 60 和 80 都会被选中。
-- pitfalls --
  * BETWEEN 是闭区间，容易误以为不含端点。
  * NOT IN 遇到集合中含 NULL 时整体结果可能为空。
-- exercises --
  (1) 用 IN 返回 id 为 1 或 5 的记录，按 id 排序。 | hint=WHERE id IN (1, 5)。
      answer: SELECT id, name FROM scores WHERE id IN (1, 5) ORDER BY id;
      expected: 1|Alice
5|Eve
  (2) 返回分数不在 60 到 80 之间的学生，按分数升序。 | hint=使用 NOT BETWEEN。
      answer: SELECT name, score FROM scores WHERE score NOT BETWEEN 60 AND 80 ORDER BY score;
      expected: Carol|59
Eve|95

#5 [basics/beginner/SQLite 3.50] LIKE 模式匹配
objective: 用 LIKE 的 % 与 _ 通配符做模糊匹配
-- setup --
CREATE TABLE users (id INTEGER PRIMARY KEY, email TEXT);
INSERT INTO users VALUES
  (1, 'alice@example.com'),
  (2, 'bob@example.com'),
  (3, 'amy@test.org'),
  (4, 'adam@example.com');
-- query --
SELECT id, email FROM users WHERE email LIKE 'a%.com' ORDER BY id;
-- expected --
1|alice@example.com
4|adam@example.com
-- explain --
LIKE 用 % 匹配任意长度（含 0）字符、_ 匹配单个字符。SQLite 3.50 中 LIKE 默认对 ASCII 大小写不敏感、对非 ASCII 敏感；可用 ESCAPE 指定转义符匹配字面 % 或 _。示例匹配 email 以 'a' 开头且以 '.com' 结尾的行，前导 'a%' 与尾部 '%.com' 组合实现首尾约束。
-- pitfalls --
  * LIKE 'a%.com' 中的 . 是普通字符，若要匹配任意单字符应使用 _。
  * 对非 ASCII 的大小写不敏感需要设置大小写折叠扩展。
-- exercises --
  (1) 返回 email 中包含 'example' 的用户 id，按 id 排序。 | hint=使用 %example%。
      answer: SELECT id FROM users WHERE email LIKE '%example%' ORDER BY id;
      expected: 1
2
4
  (2) 返回邮箱第二个字符为 'o' 的用户 id，按 id 排序。 | hint=用 _ 匹配单个字符：'_o%'。
      answer: SELECT id FROM users WHERE email LIKE '_o%' ORDER BY id;
      expected: 2

#6 [basics/beginner/SQLite 3.50] ORDER BY 排序与多列排序
objective: 用单列和多列排序并指定升降序
-- setup --
CREATE TABLE emp (id INTEGER PRIMARY KEY, name TEXT, dept TEXT, salary INTEGER);
INSERT INTO emp VALUES
  (1, 'Alice', 'sales', 9000),
  (2, 'Bob', 'sales', 12000),
  (3, 'Carol', 'engineering', 8000),
  (4, 'Dave', 'engineering', 15000);
-- query --
SELECT dept, name, salary FROM emp ORDER BY dept ASC, salary DESC;
-- expected --
engineering|Dave|15000
engineering|Carol|8000
sales|Bob|12000
sales|Alice|9000
-- explain --
ORDER BY 决定结果行顺序，默认升序 ASC，DESC 表示降序。多列排序时后面的列只在前面的列相等时才起作用。SQLite 3.50 中 NULL 在升序时排最前，可用 NULLS FIRST/LAST 显式指定。示例按部门升序、同部门内按薪资降序排列，因此 engineering 组的Dave 在 Carol 之前。
-- pitfalls --
  * 只按非唯一列排序时其余行的顺序不稳定。
  * 在 MySQL 中默认 NULL 升序排最前、降序排最后，与显式 NULLS 子句的方言不同。
-- exercises --
  (1) 按薪资从高到低返回 name 与 salary。 | hint=ORDER BY salary DESC。
      answer: SELECT name, salary FROM emp ORDER BY salary DESC;
      expected: Dave|15000
Bob|12000
Alice|9000
Carol|8000
  (2) 按薪资升序返回 name。 | hint=ORDER BY salary ASC。
      answer: SELECT name FROM emp ORDER BY salary;
      expected: Carol
Alice
Bob
Dave

#7 [basics/beginner/SQLite 3.50] LIMIT 与 OFFSET 分页
objective: 用 LIMIT/OFFSET 取前 N 行并分页
-- setup --
CREATE TABLE logs (id INTEGER PRIMARY KEY, message TEXT);
INSERT INTO logs VALUES (1,'a'),(2,'b'),(3,'c'),(4,'d'),(5,'e');
-- query --
SELECT id, message FROM logs ORDER BY id LIMIT 2 OFFSET 2;
-- expected --
3|c
4|d
-- explain --
LIMIT n 只返回前 n 行，OFFSET m 跳过前 m 行。分页必须有稳定的 ORDER BY，否则同一页可能重复或遗漏。示例按 id 升序跳过 2 行再取 2 行，即第 2 页（每页 2 行）返回 id 3、4。
-- pitfalls --
  * 没有 ORDER BY 时 LIMIT 的结果顺序不确定。
  * 大 OFFSET 会随页数增长而变慢，深分页应考虑键集分页。
-- exercises --
  (1) 取按 id 升序的前 3 行。 | hint=LIMIT 3，无 OFFSET。
      answer: SELECT id, message FROM logs ORDER BY id LIMIT 3;
      expected: 1|a
2|b
3|c
  (2) 取最后一行（跳过 4 行取 1 行），按 id 升序。 | hint=LIMIT 1 OFFSET 4。
      answer: SELECT id FROM logs ORDER BY id LIMIT 1 OFFSET 4;
      expected: 5

#8 [basics/beginner/SQLite 3.50] DISTINCT 去重
objective: 用 DISTINCT 去掉结果中的重复行
-- setup --
CREATE TABLE orders (id INTEGER PRIMARY KEY, customer TEXT, amount INTEGER);
INSERT INTO orders VALUES (1,'alice',100),(2,'alice',200),(3,'bob',150),(4,'bob',150);
-- query --
SELECT DISTINCT customer FROM orders ORDER BY customer;
-- expected --
alice
bob
-- explain --
DISTINCT 作用于整个 SELECT 列表，消除完全相同的组合行，而不是某一列。示例从orders 表取去重后的 customer 列，只有单个列的重复会被合并。DISTINCT 通常需要排序或哈希，数据量大时比 GROUP BY 有额外开销。
-- pitfalls --
  * 误以为 DISTINCT 只对紧跟的第一列生效——它针对整行组合。
  * 对含大文本列做整行去重代价高。
-- exercises --
  (1) 返回去重后的 (customer, amount) 组合，按 customer 升序、amount 升序。 | hint=SELECT DISTINCT customer, amount。
      answer: SELECT DISTINCT customer, amount FROM orders ORDER BY customer, amount;
      expected: alice|100
alice|200
bob|150
  (2) 统计有多少个不同客户（返回一个数字）。 | hint=用 COUNT(DISTINCT customer)。
      answer: SELECT COUNT(DISTINCT customer) FROM orders;
      expected: 2

#9 [basics/beginner/SQLite 3.50] 表达式与计算列
objective: 在 SELECT 中做算术运算并理解优先级
-- setup --
CREATE TABLE order_items (id INTEGER PRIMARY KEY, price REAL, qty INTEGER);
INSERT INTO order_items VALUES (1, 19.5, 3), (2, 100.0, 2);
-- query --
SELECT id, price, qty, price * qty AS subtotal FROM order_items ORDER BY id;
-- expected --
1|19.5|3|58.5
2|100.0|2|200.0
-- explain --
SELECT 列表里可以写任意表达式，算术运算符 + - * / % 中 * / % 优先级高于 + -，可用括号改变顺序。SQLite 3.50 中整数除法会截断，要保留小数需至少一个操作数为实数。示例用 (price * qty) 计算订单小计，金额保存为 REAL。
-- pitfalls --
  * 整数相除在 SQLite 中会截断，例如 7/2 = 3。
  * 除零在 SQLite 中返回 NULL，不报错。
-- exercises --
  (1) 返回 id 与 price+qty 的结果（列名 total），按 id 排序。 | hint=直接相加。
      answer: SELECT id, price + qty FROM order_items ORDER BY id;
      expected: 1|22.5
2|102.0
  (2) 返回每个商品单价加 10% 后的价格（列名 markup），按 id 排序。 | hint=price * 1.1。
      answer: SELECT id, price * 1.1 FROM order_items ORDER BY id;
      expected: 1|21.45
2|110.0

#10 [basics/beginner/SQLite 3.50] 类型亲和与 CAST
objective: 理解 SQLite 动态类型并用 CAST 显式转换
-- setup --
CREATE TABLE raw (id INTEGER PRIMARY KEY, value TEXT);
INSERT INTO raw VALUES (1, '42'), (2, '7');
-- query --
SELECT id, CAST(value AS INTEGER) + 1 AS next_int FROM raw ORDER BY id;
-- expected --
1|43
2|8
-- explain --
SQLite 是动态类型：列有“类型亲和”，但任何列都能存任意类型。CAST(x AS type) 显式转换，常用 INTEGER/TEXT/REAL。示例把文本 '42' 与文本 '7' 用 CAST 转成整数再相加得到 49；若不转，'42'+'7' 也会被 SQLite 隐式转成数字，但显式 CAST 语义更清楚。
-- pitfalls --
  * CAST 不能转换的文本会变成 0 而不是报错。
  * 类型亲和不是强制类型，写入时不会自动严格校验。
-- exercises --
  (1) 把 value 转成整数后相加求和（返回单个数字）。 | hint=SUM(CAST(value AS INTEGER))。
      answer: SELECT SUM(CAST(value AS INTEGER)) FROM raw;
      expected: 49
  (2) 返回把 id 与 value 拼成一个字符串的列 combined，按 id 排序。 | hint=用 || 拼接，必要时先 CAST id 为 TEXT。
      answer: SELECT id || '-' || value FROM raw ORDER BY id;
      expected: 1-42
2-7

#11 [basics/beginner/SQLite 3.50] 表别名与列限定名
objective: 用表别名简写并消除列名歧义
-- setup --
CREATE TABLE employees (id INTEGER PRIMARY KEY, name TEXT, salary INTEGER);
INSERT INTO employees VALUES (1,'Alice',9000),(2,'Bob',12000);
-- query --
SELECT e.name, e.salary FROM employees AS e WHERE e.salary > 10000 ORDER BY e.id;
-- expected --
Bob|12000
-- explain --
FROM 表 AS 别名 之后可以用 别名.列 限定引用。单表时别名只是简写；多表连接时限定名是消除同名列歧义的必需手段。示例给 employees 起别名 e，并用 e.name、e.salary 限定引用，效果与全名一致。
-- pitfalls --
  * 起了别名后，某些数据库不允许再用原名引用（SQLite 允许，但不可移植）。
  * 列别名与表别名不可混用，e.name 的 e 必须是表别名。
-- exercises --
  (1) 用别名 x 返回 employees 的 id 与 name，按 id 排序。 | hint=FROM employees AS x。
      answer: SELECT x.id, x.name FROM employees AS x ORDER BY x.id;
      expected: 1|Alice
2|Bob
  (2) 用别名返回薪资大于等于 9000 的 name，按 id 排序。 | hint=WHERE 中用别名限定。
      answer: SELECT e.name FROM employees AS e WHERE e.salary >= 9000 ORDER BY e.id;
      expected: Alice
Bob

#12 [basics/beginner/SQLite 3.50] SELECT 结果集与列顺序
objective: 理解结果列顺序由 SELECT 决定并可显式控制
-- setup --
CREATE TABLE t (id INTEGER PRIMARY KEY, name TEXT);
INSERT INTO t VALUES (1,'Alice'),(2,'Bob');
-- query --
SELECT name, id FROM t ORDER BY id;
-- expected --
Alice|1
Bob|2
-- explain --
结果集列的顺序、名称完全由 SELECT 列表决定：重命名表列不会影响 SELECT * 以外的显式投影，而 SELECT * 会随表结构变化。示例交换 id 与 name 的投影顺序，证明列顺序由查询而非建表决定。
-- pitfalls --
  * 依赖 SELECT * 的列顺序在表结构变更后会破坏下游。
  * 列别名与源列名相同容易混淆投影来源。
-- exercises --
  (1) 按 id 降序返回 id 与 name。 | hint=ORDER BY id DESC。
      answer: SELECT id, name FROM t ORDER BY id DESC;
      expected: 2|Bob
1|Alice
  (2) 只返回 name，按 name 降序。 | hint=ORDER BY name DESC。
      answer: SELECT name FROM t ORDER BY name DESC;
      expected: Bob
Alice

#13 [basics/beginner/SQLite 3.50] 排序中的 NULL 位置
objective: 控制 NULL 在排序结果中的位置
-- setup --
CREATE TABLE exam (id INTEGER PRIMARY KEY, name TEXT, score INTEGER);
INSERT INTO exam VALUES (1,'Alice',90),(2,'Bob',NULL),(3,'Carol',60);
-- query --
SELECT name, score FROM exam ORDER BY score ASC NULLS LAST, name;
-- expected --
Carol|60
Alice|90
Bob|
-- explain --
SQLite 3.30 起支持 ORDER BY ... NULLS FIRST/LAST。默认升序时 NULL 排最前、降序时排最后。示例显式使用 NULLS LAST 让缺失分数排在最后，因此 90、60 在前，NULL 在后。MySQL 的行为不同，需要 NULLS 子句或 ISNULL 技巧。
-- pitfalls --
  * 把 NULL 当作 0 比较会得到错误排序或过滤结果。
  * 默认 NULL 位置在 SQLite 与 MySQL 之间不一致。
-- exercises --
  (1) 把 NULL 排在前面，按分数升序返回 name 与 score（NULL 在前）。 | hint=ORDER BY score ASC NULLS FIRST。
      answer: SELECT name, score FROM exam ORDER BY score ASC NULLS FIRST;
      expected: Bob|
Carol|60
Alice|90
  (2) 返回分数不为 NULL 的学生 name，按分数降序。 | hint=WHERE score IS NOT NULL。
      answer: SELECT name FROM exam WHERE score IS NOT NULL ORDER BY score DESC;
      expected: Alice
Carol

#14 [basics/beginner/SQLite 3.50] 综合练习：单表查询
objective: 组合过滤、排序与分页完成单表查询
-- setup --
CREATE TABLE sales (id INTEGER PRIMARY KEY, region TEXT, amount INTEGER);
INSERT INTO sales VALUES
  (1,'north',150),(2,'south',90),(3,'north',300),
  (4,'east',120),(5,'south',500),(6,'east',80);
-- query --
SELECT id, region, amount FROM sales WHERE amount >= 100 ORDER BY amount DESC LIMIT 3;
-- expected --
5|south|500
3|north|300
1|north|150
-- explain --
真实查询常同时用到多个子句：WHERE 过滤、ORDER BY 排序、LIMIT 取前几名。本课用sales 表筛选出金额大于等于 100 的订单，按金额降序取前 3 条，因此返回金额最高的三条。子句顺序固定为 SELECT … FROM … WHERE … ORDER BY … LIMIT。
-- pitfalls --
  * LIMIT 在 ORDER BY 之后生效，先排序再截断。
  * WHERE 不能引用 SELECT 列表里的聚合函数。
-- exercises --
  (1) 返回所有区域去重后的列表，按字母升序。 | hint=SELECT DISTINCT region。
      answer: SELECT DISTINCT region FROM sales ORDER BY region;
      expected: east
north
south
  (2) 返回金额小于 100 的订单 id，按金额升序。 | hint=WHERE amount < 100。
      answer: SELECT id FROM sales WHERE amount < 100 ORDER BY amount;
      expected: 6
2

#15 [functions/beginner/SQLite 3.50] 字符串函数（上）：长度与截取
objective: 用 length、substr 处理字符串长度与子串
-- setup --
CREATE TABLE people (id INTEGER PRIMARY KEY, name TEXT);
INSERT INTO people VALUES (1,'Alice'),(2,'Bob'),(3,'Charlotte');

SELECT id, name, length(name) AS len, substr(name,1,1) AS initial FROM people ORDER BY id;

-- explain --
SQLite 3.50 提供 length(x) 返回字符数（BLOB 返回字节数），substr(x,start,len) 从 1 开始计位取子串。示例取姓名首字符作为缩写，并用 length 显示姓名长度；substr(name,1,1) 返回第一个字符。
-- pitfalls --
  * substr 起始下标是 1，不是 0。
  * length 对中文按字符计数，对 BLOB 按字节。
-- exercises --
  (1) 返回 name 的前 3 个字符，按 id 排序。 | hint=substr(name,1,3)。
      answer: SELECT substr(name,1,3) FROM people ORDER BY id;
      expected: Ali
Bob
Cha
  (2) 返回 name 长度大于 3 的 id，按 id 排序。 | hint=WHERE length(name) > 3。
      answer: SELECT id FROM people WHERE length(name) > 3 ORDER BY id;
      expected: 1
3

#16 [functions/beginner/SQLite 3.50] 字符串函数（下）：拼接、替换、大小写
objective: 用拼接、替换、大小写和去空白函数清洗文本
-- setup --
CREATE TABLE accounts (id INTEGER PRIMARY KEY, user_name TEXT, domain TEXT);
INSERT INTO accounts VALUES (1,'alice','example.com'),(2,'  bob  ','test.org');
-- query --
SELECT id, upper(trim(user_name)) AS uname, trim(user_name) || '@' || domain AS email FROM accounts ORDER BY id;
-- expected --
1|ALICE|alice@example.com

#33 [aggregation/beginner/SQLite 3.50] GROUP BY 分组基础
objective: 按分组列对数据分组并聚合
-- setup --
CREATE TABLE sales (id INTEGER PRIMARY KEY, region TEXT, amount INTEGER);
INSERT INTO sales VALUES (1,'north',100),(2,'south',200),(3,'north',150),(4,'south',50);
-- query --
SELECT region, SUM(amount) AS total FROM sales GROUP BY region ORDER BY region;
-- expected --
north|250
south|250
-- explain --
GROUP BY 把行按分组键分成若干组，每组产生一行聚合结果。SELECT 列表中非聚合列必须出现在 GROUP BY 中。示例按 region 分组求销售额合计，返回每个区域的总额。
-- pitfalls --
  * SELECT 里出现未分组的裸列在 SQLite 中会任取一行，语义不确定。

      answer: SELECT region, COUNT(*) FROM sales GROUP BY region ORDER BY region;
      expected: north|2
south|2
  (2) 按 region 求金额平均值，按 region 升序。 | hint=AVG(amount)。
      answer: SELECT region, AVG(amount) FROM sales GROUP BY region ORDER BY region;
      expected: north|125.0
south|125.0

#34 [aggregation/beginner/SQLite 3.50] HAVING 过滤分组
objective: 用 HAVING 在聚合后过滤分组
-- setup --
CREATE TABLE sales (id INTEGER PRIMARY KEY, region TEXT, amount INTEGER);
INSERT INTO sales VALUES (1,'north',100),(2,'south',200),(3,'north',50),(4,'east',500);
-- query --
SELECT region, SUM(amount) AS total FROM sales GROUP BY region HAVING SUM(amount) >= 200 ORDER BY region;
-- expected --
east|500
south|200
-- explain --
WHERE 在分组前过滤行，HAVING 在分组后过滤组。示例先按 region 分组求总额，再用 HAVING total >= 200 只保留总额达标的区域。HAVING 可以引用聚合结果，WHERE 不行。
-- pitfalls --
  * 把聚合条件写进 WHERE 会报错或语义错误。
  * HAVING 里列名别名在部分数据库不可用，写聚合表达式最稳妥。
-- exercises --
  (1) 保留订单数大于 1 的区域，按 region 升序。 | hint=HAVING COUNT(*) > 1。
      answer: SELECT region FROM sales GROUP BY region HAVING COUNT(*) > 1 ORDER BY region;
      expected: north
  (2) 求总额严格大于 300 的区域，按 region 升序。 | hint=HAVING SUM(amount) > 300。
      answer: SELECT region FROM sales GROUP BY region HAVING SUM(amount) > 300 ORDER BY region;
      expected: east

#35 [aggregation/beginner/SQLite 3.50] GROUP BY 多列与聚合顺序
objective: 按多列组合分组并理解分组粒度
-- setup --
CREATE TABLE sales (id INTEGER PRIMARY KEY, region TEXT, product TEXT, amount INTEGER);
INSERT INTO sales VALUES (1,'north','pen',10),(2,'north','pen',20),(3,'north','book',50),(4,'south','pen',5);
-- query --
SELECT region, product, SUM(amount) AS total FROM sales GROUP BY region, product ORDER BY region, product;
-- expected --
north|book|50
north|pen|30
south|pen|5
-- explain --
GROUP BY a, b 按 (a,b) 组合分组，分组数等于不同组合数。示例按 (region, product) 分组求和，north/pen 与 north/book 是两个不同的组。
-- pitfalls --
  * 分组列顺序影响结果行顺序，需配合 ORDER BY。
  * 组合键基数被低估会得到比预期更多的组。

      expected: north|book|1
north|pen|2
south|pen|1
  (2) 按 product 汇总总额，按总额降序。 | hint=GROUP BY product。
      answer: SELECT product, SUM(amount) FROM sales GROUP BY product ORDER BY SUM(amount) DESC;
      expected: book|50
pen|35

#36 [aggregation/beginner/SQLite 3.50] 聚合与 NULL 的交互

-- setup --
CREATE TABLE scores (id INTEGER PRIMARY KEY, name TEXT, score INTEGER);

SELECT COUNT(*) AS rows, COUNT(score) AS n, SUM(score) AS s, AVG(score) AS a FROM scores;

      expected: 2

#37 [aggregation/beginner/SQLite 3.50] GROUP_CONCAT 分组拼接
objective: 用 GROUP_CONCAT 把组内值拼成一个字符串
-- setup --
CREATE TABLE sales (id INTEGER PRIMARY KEY, region TEXT, product TEXT);
INSERT INTO sales VALUES (1,'north','pen'),(2,'north','book'),(3,'south','bag');
-- query --
SELECT region, GROUP_CONCAT(product, ',') AS products FROM sales GROUP BY region ORDER BY region;
-- expected --
north|pen,book
south|bag
-- explain --
GROUP_CONCAT(col, sep) 把组内值按 sep（默认逗号）拼接。SQLite 3.44 起支持ORDER BY 子句控制拼接顺序；本课示例保证顺序以稳定输出。示例把每个区域的产品名拼成一串。
-- pitfalls --
  * 默认分隔符是逗号，拼接文本本身含逗号会产生歧义。
  * 拼接结果长度受 string 长度限制影响。
-- exercises --
  (1) 按 region 用分号拼接 product，按 region 升序。 | hint=GROUP_CONCAT(product,';')。
      answer: SELECT region, GROUP_CONCAT(product, ';') FROM sales GROUP BY region ORDER BY region;
      expected: north|pen;book
south|bag
  (2) 统计每个区域的产品数量，按 region 升序。 | hint=COUNT(product)。
      answer: SELECT region, COUNT(product) FROM sales GROUP BY region ORDER BY region;
      expected: north|2
south|1

#38 [aggregation/intermediate/SQLite 3.50] 小计与总计（ROLLUP 思路）
objective: 用 UNION ALL 手工构造分组小计
-- setup --
CREATE TABLE sales (id INTEGER PRIMARY KEY, region TEXT, amount INTEGER);
INSERT INTO sales VALUES (1,'north',100),(2,'south',200),(3,'north',50);
-- query --
SELECT region, SUM(amount) AS total FROM sales GROUP BY region UNION ALL SELECT 'TOTAL', SUM(amount) FROM sales;
-- expected --
north|150
south|200
TOTAL|350
-- explain --
SQLite 3.50 不支持 GROUPING SETS/ROLLUP，需要 UNION ALL 拼接明细组与总计。示例先用 GROUP BY 得到各区域小计，再 UNION ALL 一行全表总计，用 'TOTAL' 标记。MySQL 8.0 原生支持 WITH ROLLUP。
-- pitfalls --
  * UNION ALL 不去重也不排序，需要外层 ORDER BY。
  * 各分支列类型应一致，否则类型比较规则会介入。
-- exercises --
  (1) 只返回全表总额（单值）。 | hint=SUM(amount)。
      answer: SELECT SUM(amount) FROM sales;
      expected: 350
  (2) 返回 north 区域总额（单值）。 | hint=WHERE region='north'。
      answer: SELECT SUM(amount) FROM sales WHERE region = 'north';
      expected: 150

#40 [aggregation/intermediate/SQLite 3.50] 综合练习：销售汇总报表
objective: 组合分组、聚合、过滤与排序生成汇总报表
-- setup --
CREATE TABLE orders (id INTEGER PRIMARY KEY, region TEXT, product TEXT, amount INTEGER);
INSERT INTO orders VALUES (1,'north','pen',10),(2,'north','pen',20),(3,'north','book',15),(4,'south','pen',40),(5,'south','book',5);
-- query --
SELECT region, product, SUM(amount) AS total FROM orders GROUP BY region, product HAVING SUM(amount) >= 30 ORDER BY total DESC, region, product;
-- expected --
south|pen|40
north|pen|30
-- explain --
本课把订单按 (region, product) 汇总，筛掉总额小于 30 的组合，再按总额降序输出。这是最常见的销售报表形态：先分组聚合，再 HAVING 过滤，最后排序。
-- pitfalls --
  * HAVING 与 WHERE 混用位置错误会改变分组前的行集。
  * 排序键不唯一时结果顺序不稳，应加次级排序键。
-- exercises --
  (1) 按 region 汇总总额，返回 region 与 total，按 region 升序。 | hint=GROUP BY region。
      answer: SELECT region, SUM(amount) AS total FROM orders GROUP BY region ORDER BY region;
      expected: north|45
south|45
  (2) 统计 product 为 pen 的订单总额（单值）。 | hint=WHERE product='pen'。
      answer: SELECT SUM(amount) FROM orders WHERE product = 'pen';
      expected: 70
