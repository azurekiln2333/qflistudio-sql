cmake_minimum_required(VERSION 3.20)
project(qfli_mysql_desktop LANGUAGES CXX)

set(QFLI_MYSQL_SOURCE "$ENV{QFLI_MYSQL_SOURCE}" CACHE PATH "Pinned Oracle source tree (RapidJSON headers)")
if(NOT QFLI_MYSQL_SOURCE)
  set(QFLI_MYSQL_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../../build/mysql-port/mysql-8.4.6")
endif()
set(QFLI_ORACLE_HOME "$ENV{QFLI_ORACLE_HOME}" CACHE PATH "Official Oracle MySQL server distribution")
find_path(RAPIDJSON_INCLUDE_DIR rapidjson/document.h
  HINTS "${QFLI_MYSQL_SOURCE}/extra/rapidjson/include" REQUIRED)
if(WIN32 OR APPLE)
  find_path(ORACLE_INCLUDE mysql.h HINTS "${QFLI_ORACLE_HOME}/include" NO_DEFAULT_PATH REQUIRED)
  find_library(ORACLE_LIBRARY NAMES libmysql mysqlclient
    HINTS "${QFLI_ORACLE_HOME}/lib" NO_DEFAULT_PATH REQUIRED)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(ORACLE REQUIRED IMPORTED_TARGET mysqlclient)
  set(ORACLE_LIBRARY PkgConfig::ORACLE)
endif()

add_library(qfli_mysql_bridge SHARED ../../src/qfli_mysql_client.cpp ../../src/qfli_mysql_ffi.cpp)
target_include_directories(qfli_mysql_bridge PRIVATE ../../src)
if(ORACLE_INCLUDE)
  target_include_directories(qfli_mysql_bridge SYSTEM PRIVATE "${ORACLE_INCLUDE}")
endif()
target_include_directories(qfli_mysql_bridge SYSTEM PRIVATE "${RAPIDJSON_INCLUDE_DIR}")
target_link_libraries(qfli_mysql_bridge PRIVATE ${ORACLE_LIBRARY})
target_compile_features(qfli_mysql_bridge PRIVATE cxx_std_17)
set_target_properties(qfli_mysql_bridge PROPERTIES CXX_VISIBILITY_PRESET hidden)
if(WIN32)
  set_target_properties(qfli_mysql_bridge PROPERTIES PREFIX "")
elseif(UNIX AND NOT APPLE)
  set_target_properties(qfli_mysql_bridge PROPERTIES INSTALL_RPATH "$ORIGIN")
endif()
add_executable(qfli_mysql_launcher launcher.cpp)
target_compile_features(qfli_mysql_launcher PRIVATE cxx_std_17)
if(MINGW)
  target_link_options(qfli_mysql_bridge PRIVATE -static)
  target_link_options(qfli_mysql_launcher PRIVATE -static -municode)
endif()
foreach(target qfli_mysql_bridge qfli_mysql_launcher)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /EHsc)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
  endif()
endforeach()
install(TARGETS qfli_mysql_bridge qfli_mysql_launcher
  RUNTIME DESTINATION qfli_mysql/bin LIBRARY DESTINATION qfli_mysql/lib)
if(QFLI_ORACLE_HOME)
  if(NOT EXISTS "${QFLI_ORACLE_HOME}/bin/mysqld${CMAKE_EXECUTABLE_SUFFIX}")
    message(FATAL_ERROR "Stage the full Oracle server distribution before packaging: ${QFLI_ORACLE_HOME}")
  endif()
  foreach(directory bin lib share docs licenses)
    if(EXISTS "${QFLI_ORACLE_HOME}/${directory}")
      install(DIRECTORY "${QFLI_ORACLE_HOME}/${directory}/"
        DESTINATION "qfli_mysql/${directory}" COMPONENT Runtime
        PATTERN "*.pdb" EXCLUDE PATTERN "*-debug.*" EXCLUDE
        PATTERN "*.a" EXCLUDE PATTERN "*.lib" EXCLUDE)
    endif()
  endforeach()
  foreach(file LICENSE README runtime.json)
    if(EXISTS "${QFLI_ORACLE_HOME}/${file}")
      install(FILES "${QFLI_ORACLE_HOME}/${file}" DESTINATION qfli_mysql COMPONENT Runtime)
    endif()
  endforeach()
  if(WIN32)
    install(FILES "${QFLI_ORACLE_HOME}/lib/libmysql.dll" DESTINATION qfli_mysql/bin COMPONENT Runtime)
  endif()
endif()
